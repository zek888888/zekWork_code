```python
@dataclass
class BattleRecord:
    """战颅将军的每一单都记录在案"""
    
    # 基础信息
    battle_id: str           # 战役编号
    timestamp: datetime      # 开战时间

    # 决策依据
    decision_context: Dict   # 当时的情报汇总
    ai_reasoning: str       # AI的推理过程
    confidence: float       # 当时置信度
    
    # 交易结果
    direction: str          # 做多/做空
    entry_price: float      # 入场价
    exit_price: float       # 出场价
    position_size: float    # 仓位大小
    
    # 战果
    pnl: float             # 盈亏金额
    pnl_percent: float     # 盈亏百分比
    result: str            # win/loss/break_even
    hold_time: timedelta   # 持仓时间
    
    # 归因分析
    exit_reason: str       # 出场原因 (止盈/止损/超时/手动)
    error_type: str        # 错误类型 (如果有)
    lessons_learned: str   # 教训总结
```

```python
class ErrorAnalyzer:
    """战颅将军的错误分析器"""
    
    ERROR_TYPES = {

        "over_confidence": {

        },
        
        "chasing_price": {
            "name": "追涨杀跌",
            "description": "情绪化追高或割肉",
            "improvement": "增加入场冷静期，禁止冲动交易"
        },
        
        "ignored_trend": {
            "name": "逆势操作",
            "description": "大趋势向下还做多",
            "improvement": "趋势之眼权重提高，逆势交易门槛提高"
        },
        
        "timing_error": {
            "name": "时机错误",
            "description": "方向对但时机不对，洗盘出局",
            "improvement": "分批建仓，降低时机敏感度"
        },
        
        "black_swan": {
            "name": "黑天鹅",
            "description": "突发消息导致极端行情",
            "improvement": "建立黑天鹅应对机制，异常波动减仓"
        }
    }
    
    def analyze_error(self, battle_record: BattleRecord) -> Dict:
        """分析单次交易错误"""
        
        error_type = self.classify_error(battle_record)
        
        analysis = {
            "error_type": error_type,
            "error_name": self.ERROR_TYPES[error_type]["name"],
            "description": self.ERROR_TYPES[error_type]["description"],
            "root_cause": self.find_root_cause(battle_record),
            "improvement": self.ERROR_TYPES[error_type]["improvement"],
            "similar_errors_count": self.count_similar_errors(error_type, days=30)
        }
        
        return analysis
    
    def classify_error(self, record: BattleRecord) -> str:
        """分类错误类型"""
        
        # 假突破判断
        if record.exit_reason == "止损" and self.is_near_resistance(record):
            return "false_breakout"
        
        # 入场过晚判断
        if record.hold_time < timedelta(minutes=15) and record.pnl < 0:
            return "late_entry"
        
        # 逆势操作判断
        if record.decision_context.get("trend") == "bearish" and record.direction == "long":
            return "ignored_trend"
        
        # 过度自信判断
        if record.confidence > 0.8 and record.pnl < 0:
            return "over_confidence"
        
        return "unknown"
```

### 参数自优化
```python
class StrategyOptimizer:
    """战颅将军的策略优化器"""
    
    def __init__(self):
        self.parameters = {
            "confidence_threshold": 0.65,      # 置信度阈值
            "position_size_base": 0.05,        # 基础仓位
            "risk_reward_min": 2.0,            # 最小风险收益比
            "max_hold_time": 8,                # 最大持仓时间(小时)
            "consecutive_loss_limit": 3        # 连续错误上限
        }
        
        self.optimization_history = []
    
    def optimize_parameters(self, battle_history: List[BattleRecord]):
        """根据历史战绩优化参数"""
        
        recent_30 = battle_history[-30:]
        recent_win_rate = len([b for b in recent_30 if b.result == "win"]) / len(recent_30)
        
        suggestions = []
        
        # 1. 优化置信度阈值
        high_conf_losses = [b for b in recent_30 if b.confidence > 0.8 and b.result == "loss"]
        if len(high_conf_losses) > 3:
            suggestions.append({
                "parameter": "confidence_threshold",
                "current": self.parameters["confidence_threshold"],
                "suggested": 0.70,
                "reason": f"最近30单中{len(high_conf_losses)}单高置信度亏损，需提高门槛"
            })
        
        # 2. 优化仓位大小
        consecutive_losses = self.get_consecutive_losses(recent_30)
        if consecutive_losses >= 3:
            suggestions.append({
                "parameter": "position_size_base",
                "current": self.parameters["position_size_base"],
                "suggested": 0.03,
                "reason": f"连续亏损{consecutive_losses}次，建议降低仓位"
            })
        
        # 3. 优化持仓时间
        timeout_losses = [b for b in recent_30 if b.exit_reason == "超时" and b.pnl < 0]
        if len(timeout_losses) > 2:
            suggestions.append({
                "parameter": "max_hold_time",
                "current": self.parameters["max_hold_time"],
                "suggested": 6,
                "reason": f"最近{len(timeout_losses)}单超时亏损，缩短持仓时间"
            })
        
        return suggestions
    
    def apply_optimization(self, suggestion: Dict):
        """应用优化建议"""
        param = suggestion["parameter"]
        old_value = self.parameters[param]
        new_value = suggestion["suggested"]
        
        self.parameters[param] = new_value
        
        self.optimization_history.append({
            "timestamp": datetime.now(),
            "parameter": param,
            "old_value": old_value,
            "new_value": new_value,
            "reason": suggestion["reason"]
        })
        
        print(f"[战颅将军] 参数优化: {param} {old_value} → {new_value}")
        print(f"  原因: {suggestion['reason']}")
```

## 🎓 学习核心 - 知识沉淀

### 成功案例库
```python
class SuccessLibrary:
    """战颅将军的成功案例库"""
    
    def __init__(self):
        self.cases = []
    
    def add_success_case(self, battle: BattleRecord):
        """添加成功案例"""
        case = {
            "pattern": self.extract_pattern(battle),
            "context": battle.decision_context,
            "reasoning": battle.ai_reasoning,
            "pnl": battle.pnl,
            "tags": self.tag_case(battle)
        }
        self.cases.append(case)
    
    def find_similar_success(self, current_context: Dict) -> Optional[Dict]:
        """寻找相似的成功案例"""
        # 基于相似度匹配历史成功案例
        similar_cases = []
        for case in self.cases:
            similarity = self.calculate_similarity(current_context, case["context"])
            if similarity > 0.8:
                similar_cases.append((case, similarity))
        
        if similar_cases:
            return max(similar_cases, key=lambda x: x[1])[0]
        return None
```

### 失败模式库
```python
class FailurePatternLibrary:
    """失败模式识别库"""
    
    def __init__(self):
        self.patterns = defaultdict(list)
    
    def learn_from_failure(self, battle: BattleRecord):
        """从失败中学习"""
        pattern = self.extract_failure_pattern(battle)
        self.patterns[pattern["type"]].append({
            "timestamp": battle.timestamp,
            "pattern": pattern,
            "lesson": battle.lessons_learned
        })
    
    def check_risk_pattern(self, current_context: Dict) -> List[str]:
        """检查当前是否存在已知风险模式"""
        warnings = []
        
        for pattern_type, cases in self.patterns.items():
            for case in cases[-5:]:  # 检查最近5次同类失败
                if self.match_pattern(current_context, case["pattern"]):
                    warnings.append({
                        "pattern": pattern_type,
                        "lesson": case["lesson"],
                        "frequency": len(cases)
                    })
        
        return warnings
```


```
王牌战颅 (Win Rate 70-80%)

传奇战颅 (Win Rate > 80%)
   └── 特点: 市场大师，能应对各种行情
```

> 从 **新兵战颅** → **熟练战颅** → **精英战颅**

- 第6个月: 胜率70%+ (晋升精英)
